📱 Analytics SDK & API Developer Guide

This engine collects 60 automated system metrics alongside custom app event telemetry, streaming data asynchronously via background workers to eliminate UI thread latency.

1. Android SDK Integration (Kotlin)

Step A: Add Internet Permission

Include the required network permission in your AndroidManifest.xml:

<uses-permission android:name="android.permission.INTERNET" />

Step B: Initialize SDK in Application Class

import android.app.Application

class MyApplication : Application() {
    override fun onCreate() {
        super.onCreate()

        // Initialize Analytics Engine once at app start
        MyAnalyticsSDK.initialize(
            context = this,
            apiKey = "live_sdk_key_9988",
            endpoint = "https://wild-bush-1cd7.trustconnect713.workers.dev/"
        )
    }
}

Step C: Track Custom Events

// Track level completion in a game
MyAnalyticsSDK.trackEvent(
    eventName = "level_completed",
    properties = mapOf(
        "level_id" to 5,
        "score" to 12500,
        "character" to "Explorer"
    )
)

// Track UI interaction or purchase
MyAnalyticsSDK.trackEvent(
    eventName = "in_app_purchase",
    properties = mapOf(
        "item_id" to "coins_pack_100",
        "price" to 0.99
    )
)

Step D: Track Automated System Metrics

// Track automated performance or startup metric
MyAnalyticsSDK.trackSystemEvent(
    eventName = "app_start",
    properties = mapOf("launch_time_ms" to 320)
)

3. Complete Kotlin SDK Client Engine

Include the MyAnalyticsSDK.kt client wrapper in your Android Studio project:

import android.content.Context
import android.os.Build
import kotlinx.coroutines.*
import org.json.JSONArray
import org.json.JSONObject
import java.net.HttpURLConnection
import java.net.URL

object MyAnalyticsSDK {

    private var appContext: Context? = null
    private var apiKey: String = ""
    private var workerUrl: String = ""
    private val sdkScope = CoroutineScope(Dispatchers.IO + SupervisorJob())

    fun initialize(context: Context, apiKey: String, endpoint: String) {
        this.appContext = context.applicationContext
        this.apiKey = apiKey
        this.workerUrl = endpoint

        // Auto-track initialization system metric
        trackSystemEvent("app_open", mapOf("sdk_version" to "1.0.0"))
    }

    fun trackEvent(eventName: String, properties: Map<String, Any> = emptyMap()) {
        sendPayload(eventName, type = "custom", properties = properties)
    }

    fun trackSystemEvent(eventName: String, properties: Map<String, Any> = emptyMap()) {
        sendPayload(eventName, type = "default", properties = properties)
    }

    private fun sendPayload(eventName: String, type: String, properties: Map<String, Any>) {
        val context = appContext ?: return
        if (workerUrl.isEmpty()) return

        sdkScope.launch {
            try {
                val url = URL(workerUrl)
                val conn = url.openConnection() as HttpURLConnection
                conn.requestMethod = "POST"
                conn.setRequestProperty("Content-Type", "application/json; charset=utf-8")
                conn.doOutput = true
                conn.connectTimeout = 5000

                val deviceObj = JSONObject().apply {
                    put("os", "Android")
                    put("os_version", Build.VERSION.RELEASE)
                    put("app_version", getAppVersion(context))
                }

                val eventObj = JSONObject().apply {
                    put("type", type)
                    put("name", eventName)
                    put("timestamp", System.currentTimeMillis() / 1000)
                    put("properties", JSONObject(properties))
                }

                val payload = JSONObject().apply {
                    put("app_id", context.packageName)
                    put("api_key", apiKey)
                    put("device", deviceObj)
                    put("events", JSONArray().apply { put(eventObj) })
                }

                conn.outputStream.use { os ->
                    val bytes = payload.toString().toByteArray(Charsets.UTF_8)
                    os.write(bytes, 0, bytes.size)
                }

                conn.responseCode
                conn.disconnect()
            } catch (e: Exception) {
                // Fail silently to safeguard application main thread runtime
            }
        }
    }

    private fun getAppVersion(context: Context): String {
        return try {
            val pInfo = context.packageManager.getPackageInfo(context.packageName, 0)
            pInfo.versionName ?: "1.0.0"
        } catch (e: Exception) {
            "1.0.0"
        }
    }
}
