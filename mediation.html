ConnectGold Mediation SDK - Official Developer Guide

Official integration manual for publishers implementing the ConnectGold Mobile Ad Mediation SDK into Android applications.

Mediation Endpoint: https://old-term-8a97.trustconnect713.workers.dev/v1/mediate
Auto-Register Endpoint: https://old-term-8a97.trustconnect713.workers.dev/v1/register

1. Executive Overview

ConnectGold operates as a Zero-Dashboard Mobile Ad Mediation Platform. Developers do not need to register accounts, manage web portals, or set up password logins. The SDK automatically registers your application, generates a unique publisher_id, and attaches it to every ad request.

2. System Requirements & ProGuard Rules

  • Minimum SDK Version: Android 5.0 (API Level 21) or higher
  • Target/Compile SDK Version: Android 14 (API Level 34) or higher
  • Java Version: Java 11 or Java 17 toolchains
  • Gradle Version: 7.0+

ProGuard / R8 Rules (proguard-rules.pro)

# Preserve ConnectGold Mediation Engine & Adapters
-keep class com.connectgold.ads.mediation.** { *; }
-keep class com.connectgold.ads.adapters.** { *; }
-keep interface com.connectgold.ads.mediation.MediationAdapter { *; }

# Preserve Google Advertising ID & Media3 ExoPlayer
-keep class com.google.android.gms.ads.identifier.** { *; }
-keep class androidx.media3.** { *; }

Add required network permissions in your AndroidManifest.xml:

<uses-permission android.permission.INTERNET />
<uses-permission android.permission.ACCESS_NETWORK_STATE />
<uses-permission android.permission.AD_ID />

4. SDK Initialization & Zero-Dashboard Auto Publisher ID

ConnectGold requires no web dashboard signups. When calling ConnectGoldSDK.initialize(...), the SDK fires a lightweight registration call to generate a unique publisher_id (e.g., cg_pub_a1b2c3d4) and outputs your profile inspection URL directly to Logcat.

Java Initialization (Java)

package com.connectgold.ads;

import android.content.Context;
import android.util.Log;
import org.json.JSONArray;
import org.json.JSONObject;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.nio.charset.StandardCharsets;
import java.util.concurrent.Executors;

public class ConnectGoldSDK {
    private static final String TAG = "ConnectGoldSDK";
    private static final String ENDPOINT_REGISTER = "https://old-term-8a97.trustconnect713.workers.dev/v1/register";
    private static final String ENDPOINT_MEDIATE = "https://old-term-8a97.trustconnect713.workers.dev/v1/mediate";

    private static String publisherId = null;
    private static boolean isInitialized = false;

    public interface WaterfallCallback {
        void onSuccess(JSONArray waterfall);
        void onError(String error);
    }

    public static void initialize(Context context, String appId) {
        if (isInitialized) return;

        Executors.newSingleThreadExecutor().execute(() -> {
            try {
                // Auto-register publisher with zero dashboard signups
                URL url = new URL(ENDPOINT_REGISTER);
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("POST");
                conn.setRequestProperty("Content-Type", "application/json");
                conn.setDoOutput(true);

                JSONObject payload = new JSONObject();
                payload.put("packageName", context.getPackageName());
                payload.put("appId", appId);

                try (OutputStream os = conn.getOutputStream()) {
                    os.write(payload.toString().getBytes(StandardCharsets.UTF_8));
                }

                if (conn.getResponseCode() == 200) {
                    String response = new String(conn.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
                    JSONObject json = new JSONObject(response);

                    publisherId = json.getString("publisher_id");
                    String profileUrl = json.getString("profile_url");

                    isInitialized = true;
                    Log.i(TAG, "ConnectGold Auto-Registered!");
                    Log.i(TAG, "Assigned Publisher ID: " + publisherId);
                    Log.i(TAG, "View Profile directly at URL: " + profileUrl);
                }
            } catch (Exception e) {
                Log.e(TAG, "Auto-registration error: " + e.getMessage());
            }
        });
    }

    public static String getPublisherId() { return publisherId; }

    public static void fetchWaterfall(Context context, String userId, String placementId, String format, JSONObject credentials, WaterfallCallback callback) {
        Executors.newSingleThreadExecutor().execute(() -> {
            try {
                URL url = new URL(ENDPOINT_MEDIATE);
                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
                conn.setRequestMethod("POST");
                conn.setRequestProperty("Content-Type", "application/json");
                conn.setDoOutput(true);

                JSONObject payload = new JSONObject();
                payload.put("userId", userId != null ? userId : "anonymous");
                payload.put("publisherId", publisherId != null ? publisherId : "anonymous_publisher"); // Attached Publisher ID
                payload.put("placementId", placementId);
                payload.put("adFormat", format.toUpperCase());
                if (credentials != null) {
                    payload.put("networkCredentials", credentials);
                }

                try (OutputStream os = conn.getOutputStream()) {
                    os.write(payload.toString().getBytes(StandardCharsets.UTF_8));
                }

                if (conn.getResponseCode() == 200) {
                    String jsonStr = new String(conn.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
                    JSONObject response = new JSONObject(jsonStr);
                    JSONArray waterfall = response.getJSONArray("waterfall_sequence");
                    callback.onSuccess(waterfall);
                } else {
                    callback.onError("Server Error: " + conn.getResponseCode());
                }
            } catch (Exception e) {
                callback.onError(e.getMessage());
            }
        });
    }
}

Kotlin Initialization (Kotlin)

package com.connectgold.ads

import android.content.Context
import android.util.Log
import org.json.JSONArray
import org.json.JSONObject
import java.io.OutputStream
import java.net.HttpURLConnection
import java.net.URL
import java.nio.charset.StandardCharsets
import java.util.concurrent.Executors

object ConnectGoldSDK {
    private const val TAG = "ConnectGoldSDK"
    private const val ENDPOINT_REGISTER = "https://old-term-8a97.trustconnect713.workers.dev/v1/register"
    private const val ENDPOINT_MEDIATE = "https://old-term-8a97.trustconnect713.workers.dev/v1/mediate"

    var publisherId: String? = null
        private set
    var isInitialized: Boolean = false
        private set

    fun initialize(context: Context, appId: String) {
        if (isInitialized) return

        Executors.newSingleThreadExecutor().execute {
            try {
                val conn = (URL(ENDPOINT_REGISTER).openConnection() as HttpURLConnection).apply {
                    requestMethod = "POST"
                    setRequestProperty("Content-Type", "application/json")
                    doOutput = true
                }

                val payload = JSONObject().apply {
                    put("packageName", context.packageName)
                    put("appId", appId)
                }

                conn.outputStream.use { os ->
                    os.write(payload.toString().toByteArray(StandardCharsets.UTF_8))
                }

                if (conn.responseCode == 200) {
                    val response = String(conn.inputStream.readAllBytes(), StandardCharsets.UTF_8)
                    val json = JSONObject(response)

                    publisherId = json.getString("publisher_id")
                    val profileUrl = json.getString("profile_url")

                    isInitialized = true
                    Log.i(TAG, "ConnectGold Auto-Registered!")
                    Log.i(TAG, "Assigned Publisher ID: $publisherId")
                    Log.i(TAG, "View Profile directly at URL: $profileUrl")
                }
            } catch (e: Exception) {
                Log.e(TAG, "Auto-registration error: ${e.message}")
            }
        }
    }

    fun fetchWaterfall(
        context: Context,
        userId: String?,
        placementId: String,
        format: String,
        credentials: JSONObject?,
        onSuccess: (JSONArray) -> Unit,
        onError: (String) -> Unit
    ) {
        Executors.newSingleThreadExecutor().execute {
            try {
                val conn = (URL(ENDPOINT_MEDIATE).openConnection() as HttpURLConnection).apply {
                    requestMethod = "POST"
                    setRequestProperty("Content-Type", "application/json")
                    doOutput = true
                }

                val payload = JSONObject().apply {
                    put("userId", userId ?: "anonymous")
                    put("publisherId", publisherId ?: "anonymous_publisher") // Attached Publisher ID
                    put("placementId", placementId)
                    put("adFormat", format.uppercase())
                    credentials?.let { put("networkCredentials", it) }
                }

                conn.outputStream.use { os ->
                    os.write(payload.toString().toByteArray(StandardCharsets.UTF_8))
                }

                if (conn.responseCode == 200) {
                    val jsonStr = String(conn.inputStream.readAllBytes(), StandardCharsets.UTF_8)
                    val response = JSONObject(jsonStr)
                    val waterfall = response.getJSONArray("waterfall_sequence")
                    onSuccess(waterfall)
                } else {
                    onError("Server Error: ${conn.responseCode}")
                }
            } catch (e: Exception) {
                onError(e.message ?: "Unknown error")
            }
        }
    }
}
